# Cette crise déclenche une vraie guerre civile vanilla : le pays rebelle est créé dynamiquement
# par le moteur du jeu. Les choix disponibles dépendent du régime en place.
country_event = {
    id = war_consequences.9
    title = war_consequences.9.t
    desc = war_consequences.9.d
    is_triggered_only = yes
    immediate = {
        set_country_flag = { flag = wc_civil_war_started days = 3650 }
        set_country_flag = { flag = wc_civil_war_cooldown days = 730 }
    }
    option = {
        name = war_consequences.9.a
        trigger = { has_government = democratic }
        ai_chance = { factor = 1 }
        start_civil_war = { ideology = communism size = 0.25 }
    }
    option = {
        name = war_consequences.9.b
        trigger = { has_government = fascism }
        ai_chance = { factor = 1 }
        start_civil_war = { ideology = democratic size = 0.25 }
    }
    option = {
        name = war_consequences.9.c
        trigger = { has_government = communism }
        ai_chance = { factor = 1 }
        start_civil_war = { ideology = democratic size = 0.25 }
    }
    option = {
        name = war_consequences.9.d
        trigger = { has_government = neutrality }
        ai_chance = { factor = 1 }
        start_civil_war = { ideology = communism size = 0.25 }
    }
    option = {
        name = war_consequences.9.e
        trigger = { NOT = { has_government = democratic has_government = fascism has_government = communism has_government = neutrality } }
        ai_chance = { factor = 0 }
        add_stability = -0.10
        add_war_support = -0.05
    }
}

# Coup d'État : change réellement le régime au pouvoir (set_politics).
country_event = {
    id = war_consequences.10
    title = war_consequences.10.t
    desc = war_consequences.10.d
    is_triggered_only = yes
    immediate = { set_country_flag = { flag = wc_coup_cooldown days = 730 } }
    option = {
        name = war_consequences.10.a
        trigger = { NOT = { has_government = fascism } }
        set_politics = { ruling_party = fascism elections_allowed = no }
        add_stability = 0.05
        add_timed_idea = { idea = wc_coup_aftermath days = 120 }
    }
    option = {
        name = war_consequences.10.b
        trigger = { NOT = { has_government = democratic } }
        set_politics = { ruling_party = democratic elections_allowed = no }
        add_stability = 0.05
        add_timed_idea = { idea = wc_coup_aftermath days = 120 }
    }
    option = {
        name = war_consequences.10.c
        add_political_power = -50
        add_stability = -0.05
        add_timed_idea = { idea = wc_military_disorder days = 90 }
    }
}
